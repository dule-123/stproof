// 替换原来的 startVerification 函数中的解密部分 async function startVerification() { // ... 前面的代码保持不变 ... try { // 重置调试信息 verificationData.debugInfo = ''; addDebugInfo('=== 开始解密过程 ==='); // 1. 读取加密文件（二进制格式） addDebugInfo(`读取加密文件: ${encFile.name} (${encFile.size} 字节)`); const encArrayBuffer = await readFileAsArrayBuffer(encFile); const encBytes = new Uint8Array(encArrayBuffer); // 检查文件大小 if (encBytes.length < 32) { // 最小: 16字节IV + 至少16字节密文 throw new Error(`加密文件太小 (${encBytes.length} 字节)，可能已损坏或格式错误`); } addDebugInfo(`加密文件总大小: ${encBytes.length} 字节`); // 2. 更安全的IV提取 addDebugInfo('提取IV...'); const ivBytes = new Uint8Array(16); for (let i = 0; i < 16; i++) { ivBytes[i] = encBytes[i]; } // 3. 提取密文（剩余部分） const ciphertextBytes = new Uint8Array(encBytes.length - 16); for (let i = 0; i < ciphertextBytes.length; i++) { ciphertextBytes[i] = encBytes[i + 16]; } addDebugInfo(`IV 长度: ${ivBytes.length} 字节`); addDebugInfo(`IV 十六进制: ${Array.from(ivBytes).map(b => b.toString(16).padStart(2, '0')).join('')}`); addDebugInfo(`密文长度: ${ciphertextBytes.length} 字节`); // 4. 读取密钥文件 addDebugInfo(`读取密钥文件: ${keyFile.name}`); const keyText = await readFileAsText(keyFile); const keyHex = keyText.trim().replace(/\s/g, ''); // 移除所有空白字符 addDebugInfo(`原始密钥内容: "${keyHex}"`); addDebugInfo(`密钥长度: ${keyHex.length} 字符`); // 5. 验证密钥格式 if (!/^[0-9a-fA-F]{64}$/.test(keyHex)) { // 尝试修复常见错误格式 let cleanedKey = keyHex; if (keyHex.length === 32) { // 可能是ASCII字符，需要转换为十六进制 cleanedKey = Array.from(keyHex).map(c => c.charCodeAt(0).toString(16).padStart(2, '0')).join(''); addDebugInfo(`检测到ASCII密钥，已转换为十六进制: ${cleanedKey}`); } if (!/^[0-9a-fA-F]{64}$/.test(cleanedKey)) { throw new Error(`密钥格式错误。应为 64 位十六进制字符，当前: ${keyHex.length} 字符，内容: ${keyHex}`); } } addDebugInfo(`密钥验证通过: 64 位十六进制`); addDebugInfo(`密钥前 16 位: ${keyHex.substring(0, 16)}...`); // 6. 将密钥转换为CryptoJS格式 const key = CryptoJS.enc.Hex.parse(keyHex); // 7. 将IV转换为CryptoJS格式（两种方法确保兼容性） let ivWordArray; try { // 方法1: 使用WordArray.create ivWordArray = CryptoJS.lib.WordArray.create(ivBytes); } catch (e) { // 方法2: 手动转换 addDebugInfo(`方法1失败，尝试手动转换IV: ${e.message}`); const ivHex = Array.from(ivBytes).map(b => b.toString(16).padStart(2, '0')).join(''); ivWordArray = CryptoJS.enc.Hex.parse(ivHex); } // 8. 将密文转换为CryptoJS格式 let ciphertextWordArray; try { ciphertextWordArray = CryptoJS.lib.WordArray.create(ciphertextBytes); } catch (e) { addDebugInfo(`密文转换失败，尝试Base64转换: ${e.message}`); // 尝试转换为Base64再解析 const base64String = btoa(String.fromCharCode.apply(null, ciphertextBytes)); ciphertextWordArray = CryptoJS.enc.Base64.parse(base64String); } addDebugInfo('开始 AES-256-CBC 解密...'); // 9. 尝试多种解密参数 let decryptedText = null; let decryptionError = null; // 方法1: 标准解密 try { addDebugInfo('尝试方法1: 标准解密'); const decrypted = CryptoJS.AES.decrypt( { ciphertext: ciphertextWordArray }, key, { iv: ivWordArray, mode: CryptoJS.mode.CBC, padding: CryptoJS.pad.Pkcs7 } ); decryptedText = decrypted.toString(CryptoJS.enc.Utf8); addDebugInfo('方法1解密成功'); } catch (err1) { decryptionError = err1.message; addDebugInfo(`方法1失败: ${err1.message}`); // 方法2: 尝试不使用IV（ECB模式，但可能性很小） try { addDebugInfo('尝试方法2: ECB模式（无IV）'); const decrypted = CryptoJS.AES.decrypt( { ciphertext: ciphertextWordArray }, key, { mode: CryptoJS.mode.ECB, padding: CryptoJS.pad.Pkcs7 } ); decryptedText = decrypted.toString(CryptoJS.enc.Utf8); addDebugInfo('方法2解密成功'); } catch (err2) { addDebugInfo(`方法2失败: ${err2.message}`); } } if (!decryptedText) { // 提供详细的错误信息 const errorDetails = `解密失败。可能原因： 1. 密钥不正确 2. 加密文件格式错误 3. 加密算法不匹配 4. IV提取错误 调试信息: - 文件大小: ${encBytes.length} 字节 - IV提取: ${ivBytes.length} 字节 - 密文大小: ${ciphertextBytes.length} 字节 - 密钥格式: ${keyHex.length === 64 ? '正确' : '错误'}`; throw new Error(errorDetails); } addDebugInfo(`解密完成，解密后文本长度: ${decryptedText.length} 字符`); addDebugInfo(`解密结果前 200 字符:`); addDebugInfo(decryptedText.substring(0, Math.min(200, decryptedText.length))); // 10. 解析 JSON addDebugInfo('开始解析 JSON...'); let jsonData; try { jsonData = JSON.parse(decryptedText); addDebugInfo('JSON 解析成功！'); } catch (jsonErr) { addDebugInfo(`JSON 解析失败: ${jsonErr.message}`); addDebugInfo('尝试清理无效字符...'); // 尝试清理多种无效字符 const cleaned = decryptedText .replace(/[\x00-\x1F\x7F-\x9F]/g, '') // 控制字符 .replace(/\s+/g, ' ') // 多个空白字符 .trim(); try { jsonData = JSON.parse(cleaned); addDebugInfo('清理后 JSON 解析成功！'); } catch (cleanErr) { // 如果还是失败，返回原始文本 addDebugInfo(`清理后仍解析失败: ${cleanErr.message}`); jsonData = { raw_text: decryptedText.substring(0, 500) + '...', parse_error: cleanErr.message }; } } // 11. 计算哈希 const hash = CryptoJS.SHA256(decryptedText).toString(); addDebugInfo(`文件哈希计算完成: ${hash.substring(0, 16)}...`); // 保存验证数据 verificationData.jsonData = jsonData; verificationData.hash = hash; // 显示验证结果 displayResult(true, jsonData, hash); } catch (error) { console.error("验证失败:", error); addDebugInfo(`验证失败: ${error.message}`); addDebugInfo(`错误堆栈: ${error.stack || '无'}`); // 显示错误信息 displayResult(false, null, null, error.message); } finally { // 恢复按钮状态 btn.disabled = false; btnText.textContent = '🔐 开始验证'; spinner.style.display = 'none'; } }
