import json from Crypto.Cipher import AES from Crypto.Util.Padding import unpad import os import glob # ============ 关键修改：指定您要解密的文件夹 ============ target_folder = r"D:\Users\d1852\OneDrive\Desktop\加密文件夹" # 切换到目标文件夹，确保所有操作都在这里进行 os.chdir(target_folder) # ===================================================== # 获取目标文件夹下所有 .enc 文件 enc_files = glob.glob("*.enc") for enc_file in enc_files: try: # 确定对应的密钥文件名 base_name = enc_file.replace('.enc', '') key_file = f"{base_name}.key.txt" # 检查密钥文件是否存在 if not os.path.exists(key_file): print(f"⚠️ 找不到密钥文件: {key_file}，跳过 {enc_file}") continue # 读取密钥 with open(key_file, 'r') as f: key_hex = f.read().strip() key = bytes.fromhex(key_hex) # 读取加密文件 with open(enc_file, 'rb') as f: # 前16字节是IV iv = f.read(16) # 剩余部分是密文 ciphertext = f.read() # 解密 cipher = AES.new(key, AES.MODE_CBC, iv=iv) decrypted_padded = cipher.decrypt(ciphertext) decrypted_bytes = unpad(decrypted_padded, AES.block_size) # 转换为字符串 decrypted_str = decrypted_bytes.decode('utf-8') # 解析JSON metadata = json.loads(decrypted_str) # 保存解密后的JSON文件 json_file = f"{base_name}_decrypted.json" with open(json_file, 'w', encoding='utf-8') as f: json.dump(metadata, f, ensure_ascii=False, indent=2) print(f"✅ 已解密: {enc_file} -> {json_file}") print(f" - 原数据摘要: {metadata.get('physical_object', {}).get('theme', '未知主题')}") print(f" - 存证时间: {metadata.get('evidence_chain', [{}])[0].get('timestamp', '未知时间')}") except Exception as e: print(f"❌ 解密失败 {enc_file}: {e}") print("\n🎉 批量解密完成！")
