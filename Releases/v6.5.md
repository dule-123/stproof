pragma solidity ^0.8.20;

/**
 * @title STProof v6.5 - Fact Asset Protocol with DAG Citation & Economic Model
 * @author STProof Team
 * @dev 基于DAG引用网络的事实资产协议，实现阶梯式引用定价、验证者机制、策展经济和手动提现体系
 * 核心改进：代码模块化、安全增强、gas优化、接口规范、错误信息标准化
 */
contract STProofV65 {
    // ==================== 常量定义 ====================
    string public constant NAME = "STProof";
    string public constant SYMBOL = "STP";
    string public constant VERSION = "6.5";
    
    // 阶梯定价常量 (单位: wei)
    uint256 public constant FEE_SOCIAL = 0.00001 ether;      // 社交级引用费用 ($0.03)
    uint256 public constant FEE_EVIDENCE = 0.0001 ether;     // 证据级引用费用 ($0.30)
    uint256 public constant FEE_COMMERCIAL = 0.001 ether;    // 商业级引用费用 ($3.00)
    
    // 引用权重常量 (定点数，1e18 = 1.0)
    uint256 public constant WEIGHT_SOCIAL = 0.1e18;          // 社交级权重
    uint256 public constant WEIGHT_EVIDENCE = 1e18;          // 证据级权重
    uint256 public constant WEIGHT_COMMERCIAL = 10e18;       // 商业级权重
    
    // 其他费用常量
    uint256 public constant ACCESS_FEE = 0.001 ether;        // 完整证据访问费用
    uint256 public constant MIN_VALIDATORS = 2;              // 最小验证者数量
    uint256 public constant MAX_EDITIONS = 10000;            // 最大铸造版数
    
    // 收益分配比例 (百分比)
    uint256 public constant CREATOR_SHARE = 70;      // 创作者分成 70%
    uint256 public constant TREASURY_SHARE = 28;     // 金库分成 28%
    uint256 public constant PROTOCOL_SHARE = 2;      // 协议缔造者分成 2%
    
    // ERC721 接口ID
    bytes4 public constant INTERFACE_ID_ERC721 = 0x80ac58cd;
    bytes4 public constant INTERFACE_ID_ERC721_METADATA = 0x5b5e139f;
    bytes4 public constant INTERFACE_ID_ERC2981 = 0x2a55205a;
    bytes4 public constant INTERFACE_ID_ERC165 = 0x01ffc9a7;
    
    // ERC721 onReceived 回调值
    bytes4 public constant ERC721_RECEIVED = 0x150b7a02;
    
    // ==================== 数据结构 ====================
    /**
     * @dev 事实资产核心结构体
     * 存储事件/证据的核心信息
     */
    struct EventProof {
        string eventName;      // 事件名称
        string cid;            // IPFS内容标识符
        address creator;       // 创建者地址
        uint256 createTime;    // 创建时间戳
        bool verified;         // 是否通过验证
        uint256 citations;     // 原始引用次数
        uint256 revenue;       // 创作者累计收益
        address[] validators;  // 验证者列表
        uint256 schemaId;      // 数据模式ID
        uint256 editionNumber; // 版本号
        uint256 totalEditions; // 总版本数
    }
    
    /**
     * @dev 引用记录结构体
     * 记录单次引用的详细信息
     */
    struct Citation {
        address citer;       // 引用者地址
        uint256 timestamp;   // 引用时间戳
        uint8 level;         // 引用级别 (1:社交, 2:证据, 3:商业)
        uint256 fee;         // 支付的费用
        uint256 weight;      // 加权权重 (定点数)
    }
    
    /**
     * @dev 验证声明结构体
     * 验证者对事实资产的验证声明
     */
    struct Attestation {
        address validator;  // 验证者地址
        string commentCID;  // 验证评论的IPFS CID
        uint256 timestamp;  // 验证时间戳
    }
    
    /**
     * @dev 策展故事线结构体
     * 用于组织和策展相关的事实资产
     */
    struct Storyline {
        uint256 id;              // 故事线ID
        string title;            // 标题
        string description;      // 描述
        address creator;         // 创建者
        uint256 createdAt;       // 创建时间
        uint256 updatedAt;       // 更新时间
        uint256[] tokenIds;      // 关联的资产ID列表
        uint256 upvoteCount;     // 点赞数
        bool isPublic;           // 是否公开
    }
    
    // ==================== 状态变量 ====================
    // 所有权
    address private _owner;
    
    // 核心配置地址
    address public treasury;          // 金库地址
    address public protocolCreator;   // 协议缔造者地址
    
    // 计数器
    uint256 private _tokenIdCounter;  // 资产ID计数器
    uint256 public storylineCounter;  // 故事线ID计数器
    
    // ERC721 核心存储
    mapping(uint256 => address) private _owners;
    mapping(address => uint256) private _balances;
    mapping(uint256 => address) private _tokenApprovals;
    mapping(address => mapping(address => bool)) private _operatorApprovals;
    
    // 事实资产存储
    mapping(uint256 => EventProof) public events;
    mapping(uint256 => string[]) public evidenceHistory;  // 证据版本历史
    mapping(uint256 => uint256) public evidenceCount;     // 证据版本数
    mapping(string => bool) public evidenceCIDMinted;     // CID防重铸
    
    // 引用网络存储
    mapping(uint256 => Citation[]) public citations;                // 引用记录
    mapping(uint256 => mapping(address => bool)) public hasCited;   // 防重复引用
    mapping(uint256 => uint256) public citationCount;               // 引用次数
    mapping(uint256 => uint256) public citationWeightedSum;         // 加权引用总和
    
    // 验证者系统
    mapping(address => bool) public validators;
    address[] public validatorList;
    
    // 验证声明存储
    mapping(uint256 => Attestation[]) public attestations;
    
    // 收益提现存储
    mapping(address => uint256) public pendingCreatorRevenue;  // 创作者待提现收益
    uint256 public pendingTreasuryRevenue;                     // 金库待提现收益
    uint256 public pendingProtocolCreatorRevenue;              // 协议缔造者待提现收益
    
    // 余额跟踪
    uint256 public treasuryBalance;    // 金库累计余额
    uint256 public creatorBalance;     // 协议缔造者累计余额
    
    // 版税存储 (EIP-2981)
    mapping(uint256 => address) private _royaltyRecipients;
    mapping(uint256 => uint256) private _royaltyRates;
    
    // 元数据存储
    string private _baseURI = "https://stproof.org/api/metadata/";
    mapping(uint256 => string) private _tokenURIs;
    
    // 故事线存储
    mapping(uint256 => Storyline) public storylines;
    mapping(address => uint256[]) public storylinesByCreator;
    mapping(uint256 => mapping(address => bool)) public storylineUpvotes;
    
    // ==================== 事件定义 ====================
    event Minted(uint256 indexed tokenId, string name, address indexed creator, uint256 schemaId, uint256 edition, uint256 total);
    event Verified(uint256 indexed tokenId, address indexed validator);
    event Cited(uint256 indexed tokenId, address indexed citer, uint256 fee, uint8 level, uint256 weight, uint256 creatorShare, uint256 treasuryShare, uint256 protocolShare);
    event ValidatorJoined(address indexed validator);
    event RevenueWithdrawn(address indexed recipient, uint256 amount, string role);
    event OwnershipTransferred(address indexed previousOwner, address indexed newOwner);
    event EvidenceAdded(uint256 indexed tokenId, string newCID, uint256 timestamp, uint256 version);
    event EditionSetMinted(string indexed cid, uint256 totalEditions, address creator);
    event AttestationAdded(uint256 indexed tokenId, address indexed validator, string commentCID, uint256 timestamp);
    event StorylineCreated(uint256 indexed storylineId, address indexed creator, string title, bool isPublic);
    event TokenAddedToStoryline(uint256 indexed storylineId, uint256 indexed tokenId);
    event StorylineUpvoted(uint256 indexed storylineId, address indexed voter);
    
    // ==================== 修饰器 ====================
    /**
     * @dev 仅所有者可调用
     */
    modifier onlyOwner() {
        require(msg.sender == _owner, "STProof: caller is not the owner");
        _;
    }
    
    /**
     * @dev 仅验证者可调用
     */
    modifier onlyValidator() {
        require(validators[msg.sender], "STProof: caller is not a validator");
        _;
    }
    
    /**
     * @dev 资产必须存在
     */
    modifier tokenExists(uint256 tokenId) {
        require(_owners[tokenId] != address(0), "STProof: token does not exist");
        _;
    }
    
    /**
     * @dev 仅资产创建者可调用
     */
    modifier onlyTokenCreator(uint256 tokenId) {
        require(events[tokenId].creator == msg.sender, "STProof: caller is not the token creator");
        _;
    }
    
    /**
     * @dev 资产必须已验证
     */
    modifier tokenVerified(uint256 tokenId) {
        require(events[tokenId].verified, "STProof: token is not verified");
        _;
    }
    
    /**
     * @dev 故事线必须存在
     */
    modifier storylineExists(uint256 storylineId) {
        require(storylines[storylineId].creator != address(0), "STProof: storyline does not exist");
        _;
    }
    
    /**
     * @dev 仅故事线创建者可调用
     */
    modifier onlyStorylineCreator(uint256 storylineId) {
        require(storylines[storylineId].creator == msg.sender, "STProof: caller is not the storyline creator");
        _;
    }
    
    // ==================== 构造函数 ====================
    /**
     * @dev 构造函数
     * @param initialOwner 初始所有者地址
     * @param _treasury 金库地址
     * @param _protocolCreator 协议缔造者地址
     */
    constructor(address initialOwner, address _treasury, address _protocolCreator) {
        require(initialOwner != address(0), "STProof: initial owner is zero address");
        require(_treasury != address(0), "STProof: treasury is zero address");
        require(_protocolCreator != address(0), "STProof: protocol creator is zero address");
        
        _owner = initialOwner;
        treasury = _treasury;
        protocolCreator = _protocolCreator;
        
        emit OwnershipTransferred(address(0), initialOwner);
    }
    
    // ==================== 所有权管理 ====================
    /**
     * @dev 获取合约所有者
     */
    function owner() public view returns (address) {
        return _owner;
    }
    
    /**
     * @dev 转移合约所有权
     * @param newOwner 新所有者地址
     */
    function transferOwnership(address newOwner) public onlyOwner {
        require(newOwner != address(0), "STProof: new owner is zero address");
        emit OwnershipTransferred(_owner, newOwner);
        _owner = newOwner;
    }
    
    // ==================== 配置管理 ====================
    /**
     * @dev 更新协议缔造者地址
     * @param newCreator 新的协议缔造者地址
     */
    function updateProtocolCreator(address newCreator) external onlyOwner {
        require(newCreator != address(0), "STProof: new creator is zero address");
        protocolCreator = newCreator;
    }
    
    /**
     * @dev 更新金库地址
     * @param newTreasury 新的金库地址
     */
    function updateTreasury(address newTreasury) external onlyOwner {
        require(newTreasury != address(0), "STProof: new treasury is zero address");
        treasury = newTreasury;
    }
    
    /**
     * @dev 设置基础URI
     * @param baseURI_ 新的基础URI
     */
    function setBaseURI(string memory baseURI_) external onlyOwner {
        _baseURI = baseURI_;
    }
    
    /**
     * @dev 设置单个资产的URI
     * @param tokenId 资产ID
     * @param tokenURI_ 新的URI
     */
    function setTokenURI(uint256 tokenId, string memory tokenURI_) external onlyOwner tokenExists(tokenId) {
        _tokenURIs[tokenId] = tokenURI_;
    }
    
    // ==================== ERC721 核心实现 ====================
    /**
     * @dev 获取合约名称
     */
    function name() public pure returns (string memory) {
        return NAME;
    }
    
    /**
     * @dev 获取合约符号
     */
    function symbol() public pure returns (string memory) {
        return SYMBOL;
    }
    
    /**
     * @dev 获取地址持有的资产数量
     * @param owner_ 持有者地址
     */
    function balanceOf(address owner_) public view returns (uint256) {
        require(owner_ != address(0), "STProof: balance query for zero address");
        return _balances[owner_];
    }
    
    /**
     * @dev 获取资产的所有者
     * @param tokenId 资产ID
     */
    function ownerOf(uint256 tokenId) public view returns (address) {
        address owner_ = _owners[tokenId];
        require(owner_ != address(0), "STProof: owner query for nonexistent token");
        return owner_;
    }
    
    /**
     * @dev 授权地址管理资产
     * @param to 被授权地址
     * @param tokenId 资产ID
     */
    function approve(address to, uint256 tokenId) public {
        address owner_ = ownerOf(tokenId);
        require(to != owner_, "STProof: approval to current owner");
        require(
            msg.sender == owner_ || isApprovedForAll(owner_, msg.sender),
            "STProof: approve caller is not owner nor approved for all"
        );
        
        _tokenApprovals[tokenId] = to;
    }
    
    /**
     * @dev 获取资产的授权地址
     * @param tokenId 资产ID
     */
    function getApproved(uint256 tokenId) public view tokenExists(tokenId) returns (address) {
        return _tokenApprovals[tokenId];
    }
    
    /**
     * @dev 批量授权地址管理所有资产
     * @param operator 操作符地址
     * @param approved 是否授权
     */
    function setApprovalForAll(address operator, bool approved) public {
        require(operator != msg.sender, "STProof: approve to caller");
        _operatorApprovals[msg.sender][operator] = approved;
    }
    
    /**
     * @dev 检查操作符是否被授权
     * @param owner_ 所有者地址
     * @param operator 操作符地址
     */
    function isApprovedForAll(address owner_, address operator) public view returns (bool) {
        return _operatorApprovals[owner_][operator];
    }
    
    /**
     * @dev 转移资产所有权
     * @param from 转出地址
     * @param to 转入地址
     * @param tokenId 资产ID
     */
    function transferFrom(address from, address to, uint256 tokenId) public {
        require(_isApprovedOrOwner(msg.sender, tokenId), "STProof: transfer caller is not owner nor approved");
        _transfer(from, to, tokenId);
    }
    
    /**
     * @dev 安全转移资产所有权（带回调检查）
     * @param from 转出地址
     * @param to 转入地址
     * @param tokenId 资产ID
     */
    function safeTransferFrom(address from, address to, uint256 tokenId) public {
        safeTransferFrom(from, to, tokenId, "");
    }
    
    /**
     * @dev 安全转移资产所有权（带回调检查和数据）
     * @param from 转出地址
     * @param to 转入地址
     * @param tokenId 资产ID
     * @param data 附加数据
     */
    function safeTransferFrom(address from, address to, uint256 tokenId, bytes memory data) public {
        require(_isApprovedOrOwner(msg.sender, tokenId), "STProof: transfer caller is not owner nor approved");
        _safeTransfer(from, to, tokenId, data);
    }
    
    /**
     * @dev 内部转移函数
     */
    function _transfer(address from, address to, uint256 tokenId) internal {
        require(ownerOf(tokenId) == from, "STProof: transfer from incorrect owner");
        require(to != address(0), "STProof: transfer to zero address");
        
        // 清除授权
        _approve(address(0), tokenId);
        
        // 更新余额
        _balances[from] -= 1;
        _balances[to] += 1;
        _owners[tokenId] = to;
    }
    
    /**
     * @dev 内部安全转移函数
     */
    function _safeTransfer(address from, address to, uint256 tokenId, bytes memory data) internal {
        _transfer(from, to, tokenId);
        require(_checkOnERC721Received(from, to, tokenId, data), "STProof: transfer to non ERC721Receiver implementer");
    }
    
    /**
     * @dev 检查地址是否被授权管理资产
     */
    function _isApprovedOrOwner(address spender, uint256 tokenId) internal view returns (bool) {
        address owner_ = ownerOf(tokenId);
        return (spender == owner_ || getApproved(tokenId) == spender || isApprovedForAll(owner_, spender));
    }
    
    /**
     * @dev 内部授权函数
     */
    function _approve(address to, uint256 tokenId) internal {
        _tokenApprovals[tokenId] = to;
    }
    
    /**
     * @dev 检查ERC721接收回调
     */
    function _checkOnERC721Received(address from, address to, uint256 tokenId, bytes memory data) internal returns (bool) {
        if (to.code.length == 0) {
            return true;
        }
        
        (bool success, bytes memory returndata) = to.call(
            abi.encodeWithSelector(
                ERC721_RECEIVED,
                msg.sender,
                from,
                tokenId,
                data
            )
        );
        
        return success && (returndata.length == 0 || abi.decode(returndata, (bytes4)) == ERC721_RECEIVED);
    }
    
    /**
     * @dev 内部安全铸造函数
     */
    function _safeMint(address to, uint256 tokenId) internal {
        _mint(to, tokenId);
        require(_checkOnERC721Received(address(0), to, tokenId, ""), "STProof: transfer to non ERC721Receiver implementer");
    }
    
    /**
     * @dev 内部铸造函数
     */
    function _mint(address to, uint256 tokenId) internal {
        require(to != address(0), "STProof: mint to zero address");
        require(_owners[tokenId] == address(0), "STProof: token already minted");
        
        _balances[to] += 1;
        _owners[tokenId] = tokenId;
    }
    
    // ==================== EIP-2981 版税实现 ====================
    /**
     * @dev 获取版税信息
     * @param tokenId 资产ID
     * @param salePrice 销售价格
     */
    function royaltyInfo(uint256 tokenId, uint256 salePrice) public view tokenExists(tokenId) returns (address receiver, uint256 royaltyAmount) {
        receiver = _royaltyRecipients[tokenId];
        royaltyAmount = (salePrice * _royaltyRates[tokenId]) / 10000;
    }
    
    /**
     * @dev 设置资产版税
     */
    function _setTokenRoyalty(uint256 tokenId, address recipient, uint256 rate) internal {
        require(rate <= 10000, "STProof: royalty rate exceeds 100%");
        _royaltyRecipients[tokenId] = recipient;
        _royaltyRates[tokenId] = rate;
    }
    
    // ==================== 资产铸造 ====================
    /**
     * @dev 基础铸造函数
     * @param to 接收地址
     * @param name_ 资产名称
     * @param cid IPFS CID
     */
    function mint(address to, string memory name_, string memory cid) external returns (uint256) {
        return mintWithSchema(to, 0, name_, cid);
    }
    
    /**
     * @dev 带模式ID的铸造函数
     * @param to 接收地址
     * @param schemaId 模式ID
     * @param name_ 资产名称
     * @param cid IPFS CID
     */
    function mintWithSchema(
        address to,
        uint256 schemaId,
        string memory name_,
        string memory cid
    ) public returns (uint256) {
        require(to != address(0), "STProof: mint to zero address");
        require(bytes(name_).length > 0, "STProof: name is required");
        require(bytes(cid).length > 0, "STProof: CID is required");
        
        uint256 tokenId = _tokenIdCounter++;
        
        // 铸造NFT
        _safeMint(to, tokenId);
        
        // 设置5%版税 (500 basis points)
        _setTokenRoyalty(tokenId, to, 500);
        
        // 初始化资产数据
        events[tokenId] = EventProof({
            eventName: name_,
            cid: cid,
            creator: to,
            createTime: block.timestamp,
            verified: false,
            citations: 0,
            revenue: 0,
            validators: new address[](0),
            schemaId: schemaId,
            editionNumber: 0,
            totalEditions: 0
        });
        
        // 初始化证据历史
        evidenceHistory[tokenId].push(cid);
        evidenceCount[tokenId] = 1;
        
        emit Minted(tokenId, name_, to, schemaId, 0, 0);
        
        return tokenId;
    }
    
    /**
     * @dev 批量铸造版本集
     * @param to 接收地址
     * @param schemaId 模式ID
     * @param eventName 资产名称
     * @param cid IPFS CID
     * @param totalEditions 总版本数
     */
    function mintEditionSet(
        address to,
        uint256 schemaId,
        string memory eventName,
        string memory cid,
        uint256 totalEditions
    ) external returns (uint256[] memory tokenIds) {
        require(to != address(0), "STProof: mint to zero address");
        require(bytes(eventName).length > 0, "STProof: name is required");
        require(bytes(cid).length > 0, "STProof: CID is required");
        require(totalEditions > 0 && totalEditions <= MAX_EDITIONS, "STProof: invalid edition count");
        require(!evidenceCIDMinted[cid], "STProof: CID already minted as edition set");
        
        evidenceCIDMinted[cid] = true;
        tokenIds = new uint256[](totalEditions);
        
        for (uint256 i = 0; i < totalEditions; i++) {
            uint256 tokenId = _tokenIdCounter++;
            
            // 铸造NFT
            _safeMint(to, tokenId);
            
            // 设置5%版税
            _setTokenRoyalty(tokenId, to, 500);
            
            // 初始化资产数据
            events[tokenId] = EventProof({
                eventName: eventName,
                cid: cid,
                creator: to,
                createTime: block.timestamp,
                verified: false,
                citations: 0,
                revenue: 0,
                validators: new address[](0),
                schemaId: schemaId,
                editionNumber: i + 1,
                totalEditions: totalEditions
            });
            
            // 初始化证据历史
            evidenceHistory[tokenId].push(cid);
            evidenceCount[tokenId] = 1;
            
            emit Minted(tokenId, eventName, to, schemaId, i + 1, totalEditions);
            tokenIds[i] = tokenId;
        }
        
        emit EditionSetMinted(cid, totalEditions, to);
    }
    
    // ==================== 动态证据管理 ====================
    /**
     * @dev 添加证据版本
     * @param tokenId 资产ID
     * @param newCID 新的证据CID
     */
    function addEvidence(uint256 tokenId, string memory newCID) external onlyTokenCreator(tokenId) tokenExists(tokenId) {
        require(bytes(newCID).length > 0, "STProof: CID is required");
        
        evidenceHistory[tokenId].push(newCID);
        evidenceCount[tokenId]++;
        
        emit EvidenceAdded(tokenId, newCID, block.timestamp, evidenceCount[tokenId]);
    }
    
    /**
     * @dev 获取证据历史
     * @param tokenId 资产ID
     */
    function getEvidenceHistory(uint256 tokenId) external view tokenExists(tokenId) returns (string[] memory) {
        return evidenceHistory[tokenId];
    }
    
    // ==================== 验证者系统 ====================
    /**
     * @dev 加入验证者列表
     */
    function joinValidator() external {
        require(!validators[msg.sender], "STProof: already a validator");
        
        validators[msg.sender] = true;
        validatorList.push(msg.sender);
        
        emit ValidatorJoined(msg.sender);
    }
    
    /**
     * @dev 验证资产
     * @param tokenId 资产ID
     */
    function verify(uint256 tokenId) external onlyValidator tokenExists(tokenId) {
        EventProof storage proof = events[tokenId];
        require(!proof.verified, "STProof: already verified");
        
        // 检查是否已验证过
        for (uint256 i = 0; i < proof.validators.length; i++) {
            require(proof.validators[i] != msg.sender, "STProof: already verified by this validator");
        }
        
        // 添加验证者
        proof.validators.push(msg.sender);
        
        // 检查是否达到最小验证者数量
        if (proof.validators.length >= MIN_VALIDATORS) {
            proof.verified = true;
        }
        
        emit Verified(tokenId, msg.sender);
    }

    // ==================== DAG引用核心功能 ====================
    /**
     * @dev 引用资产
     * @param tokenId 被引用的资产ID
     * @param level 引用级别 (1:社交, 2:证据, 3:商业)
     */
    function cite(uint256 tokenId, uint8 level) external payable tokenExists(tokenId) tokenVerified(tokenId) {
        // 验证引用级别
        require(level >= 1 && level <= 3, "STProof: invalid citation level");
        // 防止重复引用
        require(!hasCited[tokenId][msg.sender], "STProof: already cited this token");
        
        // 确定费用和权重
        uint256 fee;
        uint256 weight;
        if (level == 1) {
            fee = FEE_SOCIAL;
            weight = WEIGHT_SOCIAL;
        } else if (level == 2) {
            fee = FEE_EVIDENCE;
            weight = WEIGHT_EVIDENCE;
        } else {
            fee = FEE_COMMERCIAL;
            weight = WEIGHT_COMMERCIAL;
        }
        
        // 验证支付金额
        require(msg.value >= fee, "STProof: insufficient fee for citation level");
        
        // 更新引用数据
        EventProof storage proof = events[tokenId];
        proof.citations++;
        citationCount[tokenId]++;
        citationWeightedSum[tokenId] += weight;
        
        // 记录引用
        citations[tokenId].push(Citation({
            citer: msg.sender,
            timestamp: block.timestamp,
            level: level,
            fee: fee,
            weight: weight
        }));
        hasCited[tokenId][msg.sender] = true;
        
        // 计算收益分配
        uint256 creatorShare = (fee * CREATOR_SHARE) / 100;
        uint256 treasuryShare = (fee * TREASURY_SHARE) / 100;
        uint256 protocolShare = fee - creatorShare - treasuryShare;
        
        // 更新待提现收益
        pendingCreatorRevenue[proof.creator] += creatorShare;
        pendingTreasuryRevenue += treasuryShare;
        pendingProtocolCreatorRevenue += protocolShare;
        proof.revenue += creatorShare;
        
        // 退还多余支付
        if (msg.value > fee) {
            (bool refundSuccess, ) = payable(msg.sender).call{value: msg.value - fee}("");
            require(refundSuccess, "STProof: refund failed");
        }
        
        emit Cited(tokenId, msg.sender, fee, level, weight, creatorShare, treasuryShare, protocolShare);
    }
    
    /**
     * @dev 访问完整证据
     * @param tokenId 资产ID
     */
    function accessFullProof(uint256 tokenId) external payable tokenExists(tokenId) returns (string memory) {
        require(msg.value >= ACCESS_FEE, "STProof: insufficient access fee");
        
        EventProof storage proof = events[tokenId];
        
        // 计算收益分配
        uint256 creatorShare = (msg.value * CREATOR_SHARE) / 100;
        uint256 treasuryShare = (msg.value * TREASURY_SHARE) / 100;
        uint256 protocolShare = msg.value - creatorShare - treasuryShare;
        
        // 更新待提现收益
        pendingCreatorRevenue[proof.creator] += creatorShare;
        pendingTreasuryRevenue += treasuryShare;
        pendingProtocolCreatorRevenue += protocolShare;
        
        return proof.cid;
    }
    
    // ==================== 收益提现 ====================
    /**
     * @dev 创作者提现收益
     */
    function withdrawCreatorRevenue() external {
        uint256 amount = pendingCreatorRevenue[msg.sender];
        require(amount > 0, "STProof: no pending revenue");
        
        // 清零待提现金额（防止重入）
        pendingCreatorRevenue[msg.sender] = 0;
        
        // 转账
        (bool success, ) = payable(msg.sender).call{value: amount}("");
        require(success, "STProof: withdrawal failed");
        
        emit RevenueWithdrawn(msg.sender, amount, "creator");
    }
    
    /**
     * @dev 金库提现收益
     */
    function withdrawTreasuryRevenue() external {
        require(msg.sender == treasury, "STProof: only treasury can withdraw");
        
        uint256 amount = pendingTreasuryRevenue;
        require(amount > 0, "STProof: no pending treasury revenue");
        
        // 清零待提现金额
        pendingTreasuryRevenue = 0;
        
        // 转账
        (bool success, ) = payable(treasury).call{value: amount}("");
        require(success, "STProof: treasury withdrawal failed");
        
        // 更新累计余额
        treasuryBalance += amount;
        
        emit RevenueWithdrawn(treasury, amount, "treasury");
    }

    /**
     * @dev 获取资产元数据URI
     * @param tokenId 资产ID
     */
    function tokenURI(uint256 tokenId) public view tokenExists(tokenId) returns (string memory) {
        // 优先使用自定义URI
        if (bytes(_tokenURIs[tokenId]).length > 0) {
            return _tokenURIs[tokenId];
        }
        
        EventProof storage proof = events[tokenId];
        
        // 构建元数据JSON
        if (proof.totalEditions > 0) {
            return string(abi.encodePacked(
                _baseURI,
                '{"name":"', proof.eventName, ' #', _uint2str(proof.editionNumber), '",',
                '"attributes":[',
                '{"trait_type":"Edition","value":"', _uint2str(proof.editionNumber), '/', _uint2str(proof.totalEditions), '"},',
                '{"trait_type":"Verified","value":"', proof.verified ? "true" : "false", '"},',
                '{"trait_type":"Citations","value":', _uint2str(citationWeightedSum[tokenId] / 1e18), '},',
                '{"trait_type":"EvidenceVersions","value":', _uint2str(evidenceCount[tokenId]), '}',
                ']}'
            ));
        } else {
            return string(abi.encodePacked(
                _baseURI,
                '{"name":"', proof.eventName, '",',
                '"attributes":[',
                '{"trait_type":"Verified","value":"', proof.verified ? "true" : "false", '"},',
                '{"trait_type":"Citations","value":', _uint2str(citationWeightedSum[tokenId] / 1e18), '},',
                '{"trait_type":"EvidenceVersions","value":', _uint2str(evidenceCount[tokenId]), '}',
                ']}'
            ));
        }
    }

    /**
     * @dev 内部数字转字符串函数
     */
    function _uint2str(uint256 value) internal pure returns (string memory) {
        if (value == 0) {
            return "0";
        }
        
        uint256 temp = value;
        uint256 digits;
        
        // 计算数字位数
        while (temp != 0) {
            digits++;
            temp /= 10;
        }
        
        // 转换为字符串
        bytes memory buffer = new bytes(digits);
        while (value != 0) {
            digits--;
            buffer[digits] = bytes1(uint8(48 + value % 10));
            value /= 10;
        }
        
        return string(buffer);
    }
